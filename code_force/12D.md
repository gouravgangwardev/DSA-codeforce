## 2. D. Ball

### Approach
A lady is a probable self-murderer if another lady has:

\[
B_j > B_i,\quad I_j > I_i,\quad R_j > R_i
\]

Sort all ladies by **beauty in descending order**.

Process ladies with the same beauty as a group:

1. Query previously processed ladies.
2. Check whether there is a lady with greater intellect and greater richness.
3. Insert the current group after querying.

This prevents ladies with equal beauty from dominating each other.

Use **coordinate compression** on intellect because its values can be up to `10^9`.

A **Fenwick Tree** stores the maximum richness for intellect ranges.

### Complexity
- Time: `O(N log N)`
- Space: `O(N)`

## Solution
```
#include <bits/stdc++.h>
using namespace std;
 
struct Lady {
    int b, i, r;
};
 
struct Fenwick {
    int n;
    vector<int> bit;
 
    Fenwick(int n) : n(n), bit(n + 1, -1) {}
 
    void update(int i, int val) {
        while (i <= n) {
            bit[i] = max(bit[i], val);
            i += i & -i;
        }
    }
 
    int query(int i) {
        int ans = -1;
        while (i > 0) {
            ans = max(ans, bit[i]);
            i -= i & -i;
        }
        return ans;
    }
};
 
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
 
    int n;
    cin >> n;
 
    vector<Lady> a(n);
    vector<int> vals(n);
 
    for (int i = 0; i < n; i++)
        cin >> a[i].b;
 
    for (int i = 0; i < n; i++) {
        cin >> a[i].i;
        vals[i] = a[i].i;
    }
 
    for (int i = 0; i < n; i++)
        cin >> a[i].r;
 
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());
 
    int m = vals.size();
 
    for (auto &x : a) {
        int pos = lower_bound(vals.begin(), vals.end(), x.i) - vals.begin();
        x.i = m - pos;
    }
 
    sort(a.begin(), a.end(), [](const Lady &x, const Lady &y) {
        return x.b > y.b;
    });
 
    Fenwick fw(m);
    long long ans = 0;
 
    for (int l = 0; l < n;) {
        int r = l;
 
        while (r < n && a[r].b == a[l].b)
            r++;
 
        for (int k = l; k < r; k++) {
            if (fw.query(a[k].i - 1) > a[k].r)
                ans++;
        }
 
        for (int k = l; k < r; k++)
            fw.update(a[k].i, a[k].r);
 
        l = r;
    }
 
    cout << ans << '\n';
 
    return 0;
}
```
